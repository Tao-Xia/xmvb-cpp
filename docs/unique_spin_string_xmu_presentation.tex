\begin{frame}{Scope of This Talk}
  \begin{block}{Focus}
    This presentation focuses on the \textbf{exact} unique-spin-string
    reformulation of VBSCF determinant algebra.
  \end{block}

  \begin{itemize}
    \item The main contribution is an \textbf{exact algorithmic reorganization},
          not an approximation.
    \item The forward structure build and the analytic backward equations are
          both reformulated on unique $\alpha$ and unique $\beta$ spin spaces.
    \item Low-rank, RI, or Cholesky-based opposite-spin compression is treated
          only as \textbf{future work}.
  \end{itemize}
\end{frame}

\section{Motivation}

\begin{frame}{Why VBSCF Determinant Algebra Becomes Expensive}
  For a VB structure expansion,
  \[
  |\Phi_I\rangle = \sum_{p=1}^{\Dfull} c_{Ip}\, |D_p\rangle,
  \]
  structure-space matrix elements are assembled as
  \[
  M_{IJ}
  =
  \sum_{p=1}^{\Dfull}
  \sum_{q=1}^{\Dfull}
  c_{Ip}\, M_{pq}\, c_{Jq}.
  \]

  \begin{block}{Consequence}
    The unreduced algebra is organized around \textbf{full determinant pairs},
    so the outer topology is
    \[
    O(\Dfull^2).
    \]
  \end{block}

  \begin{itemize}
    \item The same outer structure appears in overlap, Hamiltonian, and
          analytic derivative evaluations.
    \item Improving a single determinant-pair kernel does \emph{not} remove the
          quadratic determinant-pair outer traversal.
  \end{itemize}
\end{frame}

\begin{frame}{Where the Redundancy Comes From}
  Each full determinant can be written as
  \[
  D_p = \bigl(\alpha_{i(p)}, \beta_{j(p)}\bigr),
  \]
  where $\alpha_{i(p)}$ and $\beta_{j(p)}$ are the occupied $\alpha$ and
  $\beta$ spin strings.

  \begin{block}{Key observation}
    Many full determinants share the same $\alpha$ string, the same $\beta$
    string, or both.
  \end{block}

  \begin{itemize}
    \item Therefore many same-spin determinant kernels are revisited
          repeatedly inside the full determinant-pair sweep.
    \item The bottleneck is not only a large number of determinants; it is the
          \textbf{failure to elevate shared spin strings to the outer algebraic
          representation}.
  \end{itemize}
\end{frame}

\begin{frame}{A Minimal 2-by-2 Toy Example}
  Consider
  \[
  D_1=(\alpha_1,\beta_1),\quad
  D_2=(\alpha_1,\beta_2),\quad
  D_3=(\alpha_2,\beta_1),\quad
  D_4=(\alpha_2,\beta_2).
  \]

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \[
      \begin{array}{c|cc}
           & \beta_1 & \beta_2 \\ \hline
        \alpha_1 & D_1 & D_2 \\
        \alpha_2 & D_3 & D_4
      \end{array}
      \]
      \[
      \Dfull = 4,\qquad \Na = 2,\qquad \Nb = 2.
      \]
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Counting objects}
        \[
        \text{full pairs} = 4^2 = 16
        \]
        \[
        \text{unique }\alpha\text{-pairs} = 2^2 = 4
        \]
        \[
        \text{unique }\beta\text{-pairs} = 2^2 = 4
        \]
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.4em}
  The same $(\alpha_i,\alpha_j)$ or $(\beta_k,\beta_l)$ objects are embedded in
  many different full determinant pairs.
\end{frame}

\section{Exact Reformulation}

\begin{frame}{Unique Spin Spaces}
  Let $\Na$ and $\Nb$ denote the numbers of \textbf{unique} $\alpha$ and
  \textbf{unique} $\beta$ spin strings appearing in the full determinant
  expansion.

  We define the maps
  \[
  u_{\alpha}(p)\in \{1,\dots,\Na\},
  \qquad
  u_{\beta}(p)\in \{1,\dots,\Nb\},
  \]
  which assign determinant $p$ to its unique $\alpha$ and $\beta$ IDs.

  \begin{block}{Main idea}
    Replace the full determinant list by the product space of unique spin
    strings.
  \end{block}

  \begin{itemize}
    \item The new outer objects are unique $\alpha$-pairs and unique
          $\beta$-pairs.
    \item This is an exact change of representation, not a numerical
          approximation.
  \end{itemize}
\end{frame}

\begin{frame}{Structure Coefficient Matrices}
  For each VB structure $I$, define the coefficient matrix
  \[
  \Cmat_I \in \mathbb{R}^{\Na \times \Nb},
  \]
  with elements
  \[
  \left(\Cmat_I\right)_{ab}
  =
  \sum_{p:\,u_{\alpha}(p)=a,\;u_{\beta}(p)=b}
  c_{Ip}.
  \]

  \begin{block}{What changes}
    \[
    \{c_{Ip}\}_{p=1}^{\Dfull}
    \quad \Longrightarrow \quad
    \Cmat_I \in \mathbb{R}^{\Na\times\Nb}.
    \]
  \end{block}

  \begin{itemize}
    \item Each structure is reshaped from a determinant coefficient vector into
          a matrix over unique spin spaces.
    \item Once this is done, the determinant algebra can be rewritten in terms
          of matrix products and Frobenius contractions.
  \end{itemize}
\end{frame}

\begin{frame}{Exact Matrix Form for the Overlap}
  The determinant-pair overlap factorizes exactly,
  \[
  S_{pq}
  =
  S^{\alpha}_{u_{\alpha}(p),u_{\alpha}(q)}
  S^{\beta}_{u_{\beta}(p),u_{\beta}(q)}.
  \]

  Therefore the structure-space overlap becomes
  \[
  S_{IJ}
  =
  \Frob{\Cmat_I}{\Sa \Cmat_J \Sb^{\mathsf T}}.
  \]

  \begin{block}{Interpretation}
    The overlap is assembled directly on the compressed unique-spin spaces,
    without an outer sweep over all full determinant pairs.
  \end{block}
\end{frame}

\begin{frame}{One-Electron and Same-Spin Hamiltonian}
  The exact one-electron contribution can be written as
  \[
  H^{1e}_{IJ}
  =
  \Frob{\Cmat_I}{\Honea \Cmat_J \Sb^{\mathsf T}}
  +
  \Frob{\Cmat_I}{\Sa \Cmat_J \Honeb^{\mathsf T}}.
  \]

  The exact same-spin Hamiltonian contribution is
  \[
  H^{\mathrm{ss}}_{IJ}
  =
  \Frob{\Cmat_I}{\Ha \Cmat_J \Sb^{\mathsf T}}
  +
  \Frob{\Cmat_I}{\Sa \Cmat_J \Hb^{\mathsf T}}.
  \]

  \begin{block}{Consequence}
    All reusable same-spin determinant information is lifted to dense matrices
    on the unique-spin spaces and reused by BLAS-friendly contractions.
  \end{block}
\end{frame}

\begin{frame}{Exact Opposite-Spin at the Determinant-Pair Level}
  The exact opposite-spin contribution does not factorize into a single
  $\alpha$ block times a single $\beta$ block. In packed active-pair notation,
  it can be written as
  \[
  H_{pq}^{\alpha\beta}
  =
  \sum_{P=1}^{\Npair}
  \sum_{Q=1}^{\Npair}
  U_Q^\alpha\!\bigl(u_{\alpha}(p),u_{\alpha}(q)\bigr)\,
  G_{QP}\,
  U_P^\beta\!\bigl(u_{\beta}(p),u_{\beta}(q)\bigr).
  \]

  \begin{itemize}
    \item $G_{QP}$ is the exact active-space two-electron kernel on packed
          orbital-pair indices.
    \item $U^\alpha_Q$ and $U^\beta_P$ are exact first-order cofactor
          projections regrouped from ordered same-spin determinant pairs.
  \end{itemize}
\end{frame}

\begin{frame}{Packed-Pair Channel Regrouping}
  For each packed active-pair index $P$, define
  \[
  \Ua{P}\in \mathbb{R}^{\Na\times\Na},
  \qquad
  \mathbf{U}^{\beta}_{P}\in \mathbb{R}^{\Nb\times\Nb},
  \]
  with matrix elements
  \[
  \left[\Ua{P}\right]_{aa'} = U_P^\alpha(a,a'),
  \qquad
  \left[\mathbf{U}^{\beta}_{P}\right]_{bb'} = U_P^\beta(b,b').
  \]

  Apply the exact two-electron kernel to the $\beta$ family:
  \[
  \Ub{P}
  =
  \sum_{Q=1}^{\Npair}
  G_{PQ}\, \mathbf{U}^{\beta}_{Q}.
  \]

  \begin{block}{Key point}
    The opposite-spin kernel is reorganized as a sum over packed-pair channels
    acting on matrices over the unique-spin spaces.
  \end{block}
\end{frame}

\begin{frame}{Exact Opposite-Spin Matrix Form}
  The exact opposite-spin contribution to the structure-space Hamiltonian is
  \[
  H_{IJ}^{\alpha\beta}
  =
  \sum_{P=1}^{\Npair}
  \Frob{\Cmat_I}{\Ua{P}\, \Cmat_J\, \Ub{P}^{\mathsf T}}.
  \]

  Therefore the full exact structure Hamiltonian is
  \[
  H_{IJ}
  =
  H_{IJ}^{1e}
  +
  H_{IJ}^{\mathrm{ss}}
  +
  H_{IJ}^{\alpha\beta}.
  \]

  \begin{block}{What has been eliminated}
    The exact structure build no longer requires an outer traversal over all
    full determinant pairs.
  \end{block}
\end{frame}

\begin{frame}{This Is Not a Cache Trick}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{What this is not}
        \begin{itemize}
          \item Not a scalar cache on full determinant pairs
          \item Not a small local loop optimization
          \item Not a heuristic approximation
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{What this is}
        \begin{itemize}
          \item A new unique-spin-string representation
          \item A new matrix-form structure algebra
          \item A new matrix-form backward organization
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.6em}
  \[
  \boxed{
  \text{The outer algebraic object is changed from full determinant pairs to
  unique spin spaces.}
  }
  \]
\end{frame}

\section{Analytic Gradients}

\begin{frame}{Why the Same Compression Also Applies to Gradients}
  The state-averaged target is
  \[
  E_{\mathrm{SA}} = \sum_{n=1}^{\Nstate} w_n E_n.
  \]

  For each selected state $n$, construct a unique-spin coefficient matrix
  \[
  \Cmat^{(n)} \in \mathbb{R}^{\Na\times\Nb}.
  \]

  \begin{block}{Idea}
    Instead of returning to a full determinant-pair adjoint after the
    generalized eigensolve, compress the selected-state information directly
    into unique-spin matrices and packed-pair channels.
  \end{block}
\end{frame}

\begin{frame}{Same-Spin and One-Electron Adjoint Weights}
  The same-spin / one-electron adjoint compresses to unique-spin weight
  matrices such as
  \[
  \mathbf W_{\alpha}^{(H)}
  =
  \sum_n w_n\, \Cmat^{(n)} \Sb \bigl(\Cmat^{(n)}\bigr)^{\mathsf T},
  \]
  \[
  \mathbf W_{\beta}^{(H)}
  =
  \sum_n w_n\, \bigl(\Cmat^{(n)}\bigr)^{\mathsf T} \Sa \Cmat^{(n)}.
  \]

  The overlap part is obtained by replacing $w_n$ with $-w_n E_n$:
  \[
  \mathbf W_{\alpha}^{(S)}
  =
  \sum_n (-w_n E_n)\, \Cmat^{(n)} \Sb \bigl(\Cmat^{(n)}\bigr)^{\mathsf T},
  \]
  and similarly for the $\beta$ block.

  \begin{itemize}
    \item The reverse accumulation is then performed over ordered unique-spin
          pairs rather than full determinant pairs.
    \item Forward and backward now close on the same matrix objects.
  \end{itemize}
\end{frame}

\begin{frame}{Exact Opposite-Spin Adjoint}
  The opposite-spin adjoint with respect to the packed active-space kernel can
  be written as
  \[
  \frac{\partial E}{\partial G_{QP}}
  =
  \sum_n w_n\,
  \Frob{
    \Ua{P}
  }{
    \Cmat^{(n)} \mathbf{U}^{\beta}_{Q}
    \bigl(\Cmat^{(n)}\bigr)^{\mathsf T}
  }.
  \]

  \begin{block}{Importance}
    The gradient is not only faster because the forward became faster; it is
    reorganized by the same exact unique-spin algebra.
  \end{block}

  \begin{itemize}
    \item This is why both structure build and active-space gradient evaluation
          benefit from the reformulation.
  \end{itemize}
\end{frame}

\section{Scaling}

\begin{frame}{Scaling Summary}
  \begin{block}{Original organization}
    \[
    O(\Dfull^2)
    \]
    because the outer algebra is built on full determinant pairs.
  \end{block}

  \begin{block}{Unique-spin same-spin cache}
    \[
    O(\Na^2 + \Nb^2).
    \]
  \end{block}

  \begin{block}{Exact opposite-spin structure contraction}
    \[
    O\!\left(
      \Npair \left(\Na^2\Nb + \Na\Nb^2\right)
    \right).
    \]
  \end{block}

  \begin{itemize}
    \item Same-spin determinant-pair redundancy is removed at the algorithmic
          level.
    \item The remaining frontier is the opposite-spin mixed contraction over
          packed-pair channels.
  \end{itemize}
\end{frame}

\begin{frame}{Representative Reduction in a 400-Determinant Case}
  In one representative exact benchmark,
  \[
  \Dfull = 400,\qquad \Na = 20,\qquad \Nb = 20.
  \]

  The unreduced number of same-spin determinant kernels is
  \[
  2\times \frac{\Dfull(\Dfull+1)}{2}
  =
  160{,}400.
  \]

  The corresponding number of ordered unique same-spin kernels is
  \[
  \Na^2 + \Nb^2
  =
  20^2 + 20^2
  =
  800.
  \]

  \begin{block}{Take-home message}
    This reduction is not a constant-factor tweak. It is the direct result of
    changing the algebraic representation.
  \end{block}
\end{frame}

\section{Validation and Performance}

\begin{frame}{Exactness Checks}
  The reformulation is a strict algebraic reorganization, so numerical agreement
  with the reference determinant-pair implementation is essential.

  \begin{block}{Representative exact checks}
    \[
    \max |\Delta S| \approx 2.22\times 10^{-16},
    \qquad
    \max |\Delta H| \approx 1.78\times 10^{-15}.
    \]
  \end{block}

  \begin{block}{Finite-difference checks}
    \begin{itemize}
      \item Active-space two-electron gradient errors:
            \(10^{-8}\sim 10^{-9}\)
      \item Orbital gradient errors:
            \(10^{-9}\) level
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Representative Exact Benchmark}
  For one representative exact benchmark, the determinant space contains
  \[
  \Dfull = 400,\qquad \Na = 20,\qquad \Nb = 20.
  \]

  \vspace{0.5em}
  \begin{table}
    \centering
    \begin{tabular}{lccc}
      \toprule
      Quantity & Reference & Matrix form & Speedup \\
      \midrule
      Exact structure build & 0.237 s & 0.030 s & 7.9$\times$ \\
      \bottomrule
    \end{tabular}
  \end{table}

  \begin{itemize}
    \item The structure-matrix build itself is no longer the dominant cost.
    \item The acceleration is consistent with the large reduction in repeated
          same-spin determinant work.
  \end{itemize}
\end{frame}

\begin{frame}{What Still Limits the Wall Time}
  After the same-spin reformulation, the dominant cost shifts.

  \begin{block}{Current bottlenecks}
    \begin{itemize}
      \item Construction of active-space two-electron intermediates
      \item AO-level backpropagation
      \item Exact opposite-spin mixed contraction over packed-pair channels
    \end{itemize}
  \end{block}

  \begin{itemize}
    \item This is actually a useful outcome: the determinant-pair redundancy has
          been pushed out of the critical path.
    \item The next optimization frontier is no longer same-spin reuse; it is
          opposite-spin channel compression.
  \end{itemize}
\end{frame}

\section{Outlook}

\begin{frame}{Future Directions}
  \begin{enumerate}
    \item Further compress the exact opposite-spin channel sum
    \item Explore sparse or block-sparse coefficient matrices
    \item Introduce low-rank opposite-spin representations on top of the exact
          unique-spin framework
    \item Extend the same algebra to selected-determinant or adaptive-structure
          VB/NOCI variants
  \end{enumerate}

  \vspace{0.6em}
  \begin{block}{Important separation}
    The present work is the \textbf{exact unique-spin-string reformulation}.
    Low-rank or RI/Cholesky-based compression should be viewed as the next
    layer, not as part of the current core contribution.
  \end{block}
\end{frame}

\begin{frame}{Take-Home Messages}
  \begin{enumerate}
    \item We introduced a new exact unique-spin-string representation for VBSCF
          determinant algebra.
    \item Same-spin determinant-pair redundancy is removed at the outer
          algebraic level.
    \item Forward and backward are both reformulated as matrix contractions on
          the unique-spin spaces.
    \item Numerical agreement with the reference determinant-pair formulation is
          maintained to machine precision.
    \item The remaining optimization frontier is the opposite-spin mixed
          contraction.
  \end{enumerate}

  \vspace{0.8em}
  \[
  \boxed{
  \text{This is an exact algorithmic reformulation, not a cache trick.}
  }
  \]
\end{frame}
