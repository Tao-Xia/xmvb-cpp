set(ABACUS_SOURCE_ROOT "" CACHE PATH "Pinned ABACUS checkout; only the independent spline kernel is compiled")
set(_base "${ABACUS_SOURCE_ROOT}/source/source_base")
if (NOT EXISTS "${_base}/cubic_spline.cpp" OR NOT EXISTS "${_base}/cubic_spline.h")
  message(FATAL_ERROR "ABACUS_SOURCE_ROOT must contain source/source_base/cubic_spline.{cpp,h}")
endif()
# Compile the unmodified upstream numerical kernel as an identified component.
# Do not import ABACUS CMake drivers, globals, executable or MPI configuration.
add_library(xmvb_abacus_spline STATIC "${_base}/cubic_spline.cpp")
target_include_directories(xmvb_abacus_spline SYSTEM PUBLIC "${_base}")
target_link_libraries(xmvb_abacus_spline PRIVATE ${LAPACK_LIB})
add_library(xmvb_nao STATIC radial.cpp prepare.cpp)
target_compile_features(xmvb_nao PUBLIC cxx_std_23)
target_include_directories(xmvb_nao PUBLIC ${PROJECT_SOURCE_DIR}/src ${EIGEN3_INCLUDE_DIR})
target_link_libraries(xmvb_nao PUBLIC xmvb_local_ri xmvb_quadrature PRIVATE xmvb_abacus_spline)
